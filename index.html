// ═══════════════════════════════════════════════════════ // MARKETPULSE — NEWS PAGE LOGIC v2 // Calls backend proxy — fixes NewsAPI CORS issue // ═══════════════════════════════════════════════════════ class NewsManager { constructor() { this.allArticles = []; this.category = 'all'; this.sortBy = 'date'; this.query = ''; this.page = 1; this.perPage = 9; this._debounce = null; this.init(); } async init() { await this.fetchFromBackend('all', ''); } // ── FETCH via backend proxy (fixes CORS) ───────────── async fetchFromBackend(category = 'all', q = '') { this.showSkeletons(); try { let url = `${window.apiClient.base}/api/news?category=${category}`; if (q) url += `&q=${encodeURIComponent(q)}`; const res = await fetch(url); if (!res.ok) throw new Error(`Backend returned ${res.status}`); const articles = await res.json(); if (!Array.isArray(articles)) throw new Error('Invalid response'); this.allArticles = articles; this.buildBreakingCarousel(); this.render(); } catch(e) { console.error('News fetch error:', e); this.showError(e.message); } } buildBreakingCarousel() { const top = this.allArticles.slice(0, 5); const indicators = document.getElementById('breakingIndicators'); const inner = document.getElementById('breakingInner'); if (!indicators || !inner || !top.length) return; indicators.innerHTML = top.map((_, i) => `` ).join(''); inner.innerHTML = top.map((a, i) => `
${a.image ? `` : ''}
Breaking
${a.title}
${a.source} · ${Fmt.timeAgo(a.published_at)}
`).join(''); } getFiltered() { let articles = [...this.allArticles]; // Client-side search filter on top of what backend returned if (this.query) { const q = this.query.toLowerCase(); articles = articles.filter(a => (a.title || '').toLowerCase().includes(q) || (a.description || '').toLowerCase().includes(q) || (a.source || '').toLowerCase().includes(q) ); } if (this.sortBy === 'source') { articles.sort((a, b) => (a.source || '').localeCompare(b.source || '')); } else { articles.sort((a, b) => new Date(b.published_at) - new Date(a.published_at)); } return articles; } render() { const articles = this.getFiltered(); const paginated = articles.slice(0, this.page * this.perPage); const countEl = document.getElementById('articleCount'); if (countEl) countEl.textContent = articles.length; const loadMore = document.getElementById('loadMoreWrap'); if (loadMore) loadMore.style.display = articles.length > paginated.length ? 'block' : 'none'; const grid = document.getElementById('newsGrid'); if (!grid) return; if (!articles.length) { grid.innerHTML = `
🔍

No articles found

Try a different search term or category.

`; return; } grid.innerHTML = paginated.map(a => `
${a.image ? `` : '
📰
'}
${a.source || 'Unknown'} ${Fmt.timeAgo(a.published_at)}
${a.title || 'Untitled'}
${a.description || ''}
Read article →
`).join(''); window.scrollAnimator?.refresh(); } // Category — fetch new data from backend setCategory(cat, el) { document.querySelectorAll('.filter-chips .chip').forEach(c => c.classList.remove('active')); el.classList.add('active'); this.category = cat; this.page = 1; this.query = ''; const searchEl = document.getElementById('newsSearch'); if (searchEl) searchEl.value = ''; this.fetchFromBackend(cat, ''); } // Sort — client-side only setSort(val) { this.sortBy = val; this.render(); } // Search — debounced, calls backend for new results search(q) { this.query = q; this.page = 1; // Update clear button visibility const bar = document.querySelector('.search-bar'); if (bar) bar.classList.toggle('has-value', q.length > 0); clearTimeout(this._debounce); this._debounce = setTimeout(() => { if (q.length === 0) { this.fetchFromBackend(this.category, ''); } else if (q.length >= 2) { this.fetchFromBackend('all', q); } }, 400); } clearSearch() { this.query = ''; const el = document.getElementById('newsSearch'); if (el) { el.value = ''; el.focus(); } const bar = document.querySelector('.search-bar'); if (bar) bar.classList.remove('has-value'); this.fetchFromBackend(this.category, ''); } loadMore() { this.page++; this.render(); Toast.show('More articles loaded'); } showSkeletons() { const grid = document.getElementById('newsGrid'); if (!grid) return; grid.innerHTML = Array(9).fill(`
`).join(''); } showError(msg) { const grid = document.getElementById('newsGrid'); if (!grid) return; grid.innerHTML = `
⚠️

News unavailable

Make sure the backend is running: uvicorn main:app --port 8000
${msg ? `${msg}` : ''}

`; } } document.addEventListener('DOMContentLoaded', () => { window.newsManager = new NewsManager(); }); function setNewsCategory(cat, el) { window.newsManager?.setCategory(cat, el); } function setNewsSort(val) { window.newsManager?.setSort(val); } function searchNews(q) { window.newsManager?.search(q); } function clearNewsSearch() { window.newsManager?.clearSearch(); } function loadMoreNews() { window.newsManager?.loadMore(); }
