def cache_valid(path, minutes=15):
    if not os.path.exists(path): return False
    return (datetime.now().timestamp() - os.path.getmtime(path)) / 60 < minutes

def read_cache(path):
    with open(path) as f: return json.load(f)

def write_cache(path, data):
    with open(path, "w") as f: json.dump(data, f)

@app.get("/api/watchlist")
async def get_watchlist():
    path = f"{CACHE_DIR}/watchlist.json"
    if cache_valid(path, 5): return read_cache(path)

    result = {"crypto": [], "stocks": [], "commodities": []}
    async with httpx.AsyncClient(timeout=15) as client:

        # CRYPTO
        try:
            r = await client.get("https://api.coingecko.com/api/v3/coins/markets", params={
                "vs_currency": "usd",
                "ids": "bitcoin,ethereum,solana,cardano,ripple,binancecoin",
                "sparkline": True, "price_change_percentage": "24h,7d"
            })
            for c in r.json():
                result["crypto"].append({
                    "symbol": c["symbol"].upper(), "name": c["name"],
                    "price": c["current_price"], "market_cap": c["market_cap"],
                    "change_24h": round(c.get("price_change_percentage_24h") or 0, 2),
                    "change_7d":  round(c.get("price_change_percentage_7d_in_currency") or 0, 2),
                    "volume": c["total_volume"], "image": c["image"],
                    "sparkline": (c.get("sparkline_in_7d") or {}).get("price", [])
                })
        except Exception as e: print(f"Crypto error: {e}")

        # STOCKS
        for sym in ["AAPL", "TSLA", "MSFT", "NVDA", "AMZN", "GOOGL"]:
            try:
                r = await client.get(
                    f"https://query1.finance.yahoo.com/v8/finance/chart/{sym}",
                    params={"interval": "1d", "range": "7d"},
                    headers={"User-Agent": "Mozilla/5.0"}
                )
                d    = r.json()
                meta = d["chart"]["result"][0]["meta"]
                closes = [x for x in d["chart"]["result"][0]["indicators"]["quote"][0]["close"] if x]
                result["stocks"].append({
                    "symbol": sym, "name": meta.get("shortName", sym),
                    "price": round(meta["regularMarketPrice"], 2),
                    "change_24h": round(((closes[-1]-closes[-2])/closes[-2])*100, 2) if len(closes)>=2 else 0,
                    "change_7d":  round(((closes[-1]-closes[0])/closes[0])*100, 2)  if len(closes)>=2 else 0,
                    "volume": meta.get("regularMarketVolume", 0), "market_cap": 0,
                    "sparkline": closes
                })
            except Exception as e: print(f"Stock error {sym}: {e}")

        # COMMODITIES
        for sym, name in {"GC=F":"Gold","SI=F":"Silver","CL=F":"Crude Oil","NG=F":"Natural Gas","ZW=F":"Wheat"}.items():
            try:
                r = await client.get(
                    f"https://query1.finance.yahoo.com/v8/finance/chart/{sym}",
                    params={"interval": "1d", "range": "7d"},
                    headers={"User-Agent": "Mozilla/5.0"}
                )
                d    = r.json()
                meta = d["chart"]["result"][0]["meta"]
                closes = [x for x in d["chart"]["result"][0]["indicators"]["quote"][0]["close"] if x]
                result["commodities"].append({
                    "symbol": sym.replace("=F",""), "name": name,
                    "price": round(meta["regularMarketPrice"], 2),
                    "change_24h": round(((closes[-1]-closes[-2])/closes[-2])*100, 2) if len(closes)>=2 else 0,
                    "change_7d":  round(((closes[-1]-closes[0])/closes[0])*100, 2)  if len(closes)>=2 else 0,
                    "volume": meta.get("regularMarketVolume", 0), "market_cap": 0,
                    "sparkline": closes
                })
            except Exception as e: print(f"Commodity error {sym}: {e}")

    write_cache(path, result)
    return result

# Financial sources only — prevents sports/entertainment bleed-through
FINANCIAL_DOMAINS = (
    "bloomberg.com,reuters.com,cnbc.com,ft.com,wsj.com,"
    "forbes.com,businessinsider.com,marketwatch.com,investing.com,"
    "coindesk.com,cointelegraph.com,decrypt.co,theblock.co,"
    "seekingalpha.com,barrons.com,economist.com,finance.yahoo.com"
)

CATEGORY_QUERIES = {
    "all":         [
        '(bitcoin OR ethereum OR crypto) AND (price OR market OR trading)',
        '(stock market OR nasdaq OR S&P500) AND (trading OR earnings OR rally)',
        '(federal reserve OR Fed) AND (rate OR inflation OR monetary)',
        '(gold OR crude oil) AND (price OR market OR commodity)',
    ],
    "crypto":      ['(bitcoin OR ethereum OR solana OR crypto) AND (price OR market OR defi OR blockchain)'],
    "stocks":      ['(stock market OR nasdaq OR S&P500 OR earnings OR IPO) AND (trading OR investor OR rally)'],
    "macro":       ['(federal reserve OR ECB OR interest rate OR inflation OR GDP OR CPI OR recession)'],
    "commodities": ['(gold price OR crude oil OR silver OR natural gas OR wheat) AND (commodity OR market OR trading)'],
    "tech":        ['(nvidia OR artificial intelligence OR semiconductor OR tech stocks) AND (market OR earnings OR AI)'],
}

@app.get("/api/news")
async def get_news(q: str = "", category: str = "all"):
    safe_q   = q[:30].replace(' ','_').replace('/','').replace('\','')
    cache_key = f"{CACHE_DIR}/news_{category}_{safe_q}.json"
    if cache_valid(cache_key, 20): return read_cache(cache_key)

    queries = [q] if q else CATEGORY_QUERIES.get(category, CATEGORY_QUERIES["all"])

    articles, seen = [], set()
    async with httpx.AsyncClient(timeout=20) as client:
        for query in queries[:4]:
            try:
                params = {
                    "q":        query,
                    "language": "en",
                    "sortBy":   "publishedAt",
                    "pageSize": 10,
                    "apiKey":   NEWSAPI_KEY,
                }
                # For generic queries add domain filter to avoid sports/entertainment
                if not q:
                    params["domains"] = FINANCIAL_DOMAINS

                r    = await client.get("https://newsapi.org/v2/everything", params=params)
                data = r.json()

                if data.get("status") != "ok":
                    print(f"NewsAPI error: {data.get('message')}")
                    continue

                for a in data.get("articles", []):
                    title = a.get("title","")
                    url   = a.get("url","")
                    if (not url or not title
                        or "[Removed]" in title
                        or url in seen
                        or len(title) < 10):
                        continue
                    seen.add(url)
                    articles.append({
                        "title":        title,
                        "description":  a.get("description","") or "",
                        "url":          url,
                        "image":        a.get("urlToImage"),
                        "source":       a.get("source",{}).get("name","Unknown"),
                        "published_at": a.get("publishedAt",""),
                    })
            except Exception as e:
                print(f"News fetch error: {e}")

    articles.sort(key=lambda x: x.get("published_at",""), reverse=True)
    result = articles[:24]
    if result: write_cache(cache_key, result)
    return result

# ── AI MARKET MOOD ────────────────────────────────────────
@app.get("/api/mood")
async def get_mood():
    path = f"{CACHE_DIR}/mood.json"
    if cache_valid(path, 120): return read_cache(path)

    if not ANTHROPIC_KEY:
        return {"sentiment": "Mixed", "analysis": "Add your Anthropic API key as environment variable ANTHROPIC_KEY to enable AI market mood analysis.", "generated_at": datetime.now().isoformat()}

    try:
        watchlist = await get_watchlist()
        news      = await get_news()

        crypto_s    = ", ".join([f"{c['symbol']} {'+' if c['change_24h']>0 else ''}{c['change_24h']}%" for c in watchlist.get("crypto",[])])
        stocks_s    = ", ".join([f"{s['symbol']} {'+' if s['change_24h']>0 else ''}{s['change_24h']}%" for s in watchlist.get("stocks",[])])
        commodity_s = ", ".join([f"{c['name']} {'+' if c['change_24h']>0 else ''}{c['change_24h']}%" for c in watchlist.get("commodities",[])])
        headlines   = " | ".join([a["title"] for a in news[:5]])

        prompt = f"""You are a professional market analyst. Analyze the following real-time data and provide a concise market mood summary.

Crypto (24h): {crypto_s}
Stocks (24h): {stocks_s}
Commodities (24h): {commodity_s}
Top headlines: {headlines}

Respond with:
Line 1: One word sentiment only — Bullish, Bearish, Neutral, or Mixed
Line 2-4: 2-3 sentences of specific, actionable analysis mentioning key movers and macro context.

Be direct and professional."""

        async with httpx.AsyncClient(timeout=30) as client:
            r = await client.post("https://api.anthropic.com/v1/messages",
                headers={"x-api-key": ANTHROPIC_KEY, "anthropic-version": "2023-06-01", "content-type": "application/json"},
                json={"model": "claude-haiku-4-5-20251001", "max_tokens": 250, "messages": [{"role": "user", "content": prompt}]}
            )
            text  = r.json()["content"][0]["text"].strip()
            lines = text.split("\n")
            sentiment = "Mixed"
            for word in ["Bullish", "Bearish", "Neutral", "Mixed"]:
                if lines[0].strip().lower() == word.lower():
                    sentiment = word
                    break
            analysis = " ".join(lines[1:]).strip() or text

        result = {"sentiment": sentiment, "analysis": analysis, "generated_at": datetime.now().isoformat()}
        write_cache(path, result)
        return result

    except Exception as e:
        print(f"Mood error: {e}")
        return {"sentiment": "Mixed", "analysis": "Market mood analysis temporarily unavailable.", "generated_at": datetime.now().isoformat()}

@app.get("/api/clear-cache")
async def clear_cache():
    """Clears all cached data — useful after restarting"""
    import glob
    files = glob.glob(f"{CACHE_DIR}/*.json")
    for f in files:
        try: os.remove(f)
        except: pass
    return {"cleared": len(files), "message": "Cache cleared"}
