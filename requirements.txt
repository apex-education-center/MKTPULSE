def is_cache_valid(path, minutes=15):
    if not os.path.exists(path):
        return False
    age = (datetime.now().timestamp() - os.path.getmtime(path)) / 60
    return age < minutes

def read_cache(path):
    with open(path) as f:
        return json.load(f)

def write_cache(path, data):
    with open(path, "w") as f:
        json.dump(data, f)

# ─── WATCHLIST ────────────────────────────────────────────────────────────────

@app.get("/api/watchlist")
async def get_watchlist():
    cache_path = f"{CACHE_DIR}/watchlist.json"
    if is_cache_valid(cache_path, minutes=5):
        return read_cache(cache_path)

    result = {"crypto": [], "stocks": [], "commodities": []}

    async with httpx.AsyncClient(timeout=15) as client:
        # CRYPTO — CoinGecko (free, no key)
        try:
            r = await client.get(
                "https://api.coingecko.com/api/v3/coins/markets",
                params={
                    "vs_currency": "usd",
                    "ids": "bitcoin,ethereum,solana,cardano,ripple,binancecoin",
                    "sparkline": True,
                    "price_change_percentage": "24h,7d"
                }
            )
            for c in r.json():
                result["crypto"].append({
                    "symbol": c["symbol"].upper(),
                    "name": c["name"],
                    "price": c["current_price"],
                    "change_24h": round(c.get("price_change_percentage_24h") or 0, 2),
                    "change_7d": round(c.get("price_change_percentage_7d_in_currency") or 0, 2),
                    "market_cap": c["market_cap"],
                    "volume": c["total_volume"],
                    "sparkline": (c.get("sparkline_in_7d") or {}).get("price", []),
                    "image": c["image"]
                })
        except Exception as e:
            print(f"Crypto fetch error: {e}")

        # STOCKS — Yahoo Finance (free, no key via yfinance proxy)
        stock_symbols = ["AAPL", "TSLA", "MSFT", "NVDA", "AMZN", "GOOGL"]
        for sym in stock_symbols:
            try:
                r = await client.get(
                    f"https://query1.finance.yahoo.com/v8/finance/chart/{sym}",
                    params={"interval": "1d", "range": "7d"},
                    headers={"User-Agent": "Mozilla/5.0"}
                )
                data = r.json()
                meta = data["chart"]["result"][0]["meta"]
                closes = data["chart"]["result"][0]["indicators"]["quote"][0]["close"]
                closes = [x for x in closes if x is not None]
                change_24h = round(((closes[-1] - closes[-2]) / closes[-2]) * 100, 2) if len(closes) >= 2 else 0
                change_7d = round(((closes[-1] - closes[0]) / closes[0]) * 100, 2) if len(closes) >= 2 else 0
                result["stocks"].append({
                    "symbol": sym,
                    "name": meta.get("shortName", sym),
                    "price": round(meta["regularMarketPrice"], 2),
                    "change_24h": change_24h,
                    "change_7d": change_7d,
                    "volume": meta.get("regularMarketVolume", 0),
                    "sparkline": closes
                })
            except Exception as e:
                print(f"Stock fetch error {sym}: {e}")

        # COMMODITIES — Yahoo Finance
        commodity_map = {
            "GC=F": "Gold",
            "SI=F": "Silver",
            "CL=F": "Crude Oil (WTI)",
            "NG=F": "Natural Gas",
            "ZW=F": "Wheat"
        }
        for sym, name in commodity_map.items():
            try:
                r = await client.get(
                    f"https://query1.finance.yahoo.com/v8/finance/chart/{sym}",
                    params={"interval": "1d", "range": "7d"},
                    headers={"User-Agent": "Mozilla/5.0"}
                )
                data = r.json()
                meta = data["chart"]["result"][0]["meta"]
                closes = data["chart"]["result"][0]["indicators"]["quote"][0]["close"]
                closes = [x for x in closes if x is not None]
                change_24h = round(((closes[-1] - closes[-2]) / closes[-2]) * 100, 2) if len(closes) >= 2 else 0
                change_7d = round(((closes[-1] - closes[0]) / closes[0]) * 100, 2) if len(closes) >= 2 else 0
                result["commodities"].append({
                    "symbol": sym.replace("=F", ""),
                    "name": name,
                    "price": round(meta["regularMarketPrice"], 2),
                    "change_24h": change_24h,
                    "change_7d": change_7d,
                    "volume": meta.get("regularMarketVolume", 0),
                    "sparkline": closes,
                    "unit": "USD/oz" if sym in ["GC=F", "SI=F"] else "USD/bbl" if sym == "CL=F" else "USD"
                })
            except Exception as e:
                print(f"Commodity fetch error {sym}: {e}")

    write_cache(cache_path, result)
    return result


# ─── NEWS ─────────────────────────────────────────────────────────────────────

@app.get("/api/news")
async def get_news():
    cache_path = f"{CACHE_DIR}/news.json"
    if is_cache_valid(cache_path, minutes=30):
        return read_cache(cache_path)

    articles = []
    queries = ["financial markets", "cryptocurrency", "oil commodities", "federal reserve", "stock market"]

    async with httpx.AsyncClient(timeout=15) as client:
        for q in queries[:3]:
            try:
                r = await client.get(
                    "https://newsapi.org/v2/everything",
                    params={
                        "q": q,
                        "language": "en",
                        "sortBy": "publishedAt",
                        "pageSize": 5,
                        "apiKey": NEWSAPI_KEY
                    }
                )
                for a in r.json().get("articles", [])[:3]:
                    if a.get("title") and a.get("url"):
                        articles.append({
                            "title": a["title"],
                            "description": a.get("description", ""),
                            "url": a["url"],
                            "image": a.get("urlToImage"),
                            "source": a.get("source", {}).get("name", "Unknown"),
                            "published_at": a.get("publishedAt", "")
                        })
            except Exception as e:
                print(f"News fetch error: {e}")

    # Deduplicate by URL
    seen = set()
    unique = []
    for a in articles:
        if a["url"] not in seen:
            seen.add(a["url"])
            unique.append(a)

    write_cache(cache_path, unique[:12])
    return unique[:12]

# ─── AI MARKET MOOD ───────────────────────────────────────────────────────────

@app.get("/api/mood")
async def get_market_mood():
    cache_path = f"{CACHE_DIR}/mood.json"
    if is_cache_valid(cache_path, minutes=120):
        return read_cache(cache_path)

    try:
        watchlist = await get_watchlist()
        news = await get_news()

        crypto_summary = ", ".join([
            f"{c['symbol']} {'+' if c['change_24h'] > 0 else ''}{c['change_24h']}%"
            for c in watchlist.get("crypto", [])
        ])
        stock_summary = ", ".join([
            f"{s['symbol']} {'+' if s['change_24h'] > 0 else ''}{s['change_24h']}%"
            for s in watchlist.get("stocks", [])
        ])
        commodity_summary = ", ".join([
            f"{c['name']} {'+' if c['change_24h'] > 0 else ''}{c['change_24h']}%"
            for c in watchlist.get("commodities", [])
        ])
        news_headlines = " | ".join([a["title"] for a in news[:5]])

        prompt = f"""You are a professional market analyst. Based on the following real-time market data, provide a brief, insightful market mood summary in 3-4 sentences. Be specific, mention key movers, and give a clear overall sentiment (Bullish / Bearish / Neutral / Mixed).

Crypto (24h): {crypto_summary}
Stocks (24h): {stock_summary}
Commodities (24h): {commodity_summary}
Latest headlines: {news_headlines}

Respond with:
1. Overall sentiment label (one word: Bullish/Bearish/Neutral/Mixed)
2. 3-4 sentence analysis
Keep it professional, specific, and actionable."""

        async with httpx.AsyncClient(timeout=30) as client:
            r = await client.post(
                "https://api.anthropic.com/v1/messages",
                headers={
                    "x-api-key": ANTHROPIC_KEY,
                    "anthropic-version": "2023-06-01",
                    "content-type": "application/json"
                },
                json={
                    "model": "claude-haiku-4-5-20251001",
                    "max_tokens": 300,
                    "messages": [{"role": "user", "content": prompt}]
                }
            )
            text = r.json()["content"][0]["text"]
            lines = text.strip().split("\n")
            sentiment = "Mixed"
            for word in ["Bullish", "Bearish", "Neutral", "Mixed"]:
                if word.lower() in text.lower()[:50]:
                    sentiment = word
                    break

            result = {"sentiment": sentiment, "analysis": text, "generated_at": datetime.now().isoformat()}
            write_cache(cache_path, result)
            return result

    except Exception as e:
        print(f"AI mood error: {e}")
        return {
            "sentiment": "Mixed",
            "analysis": "Market mood analysis temporarily unavailable. Please check back shortly.",
            "generated_at": datetime.now().isoformat()
        }
