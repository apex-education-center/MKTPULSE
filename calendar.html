// ═══════════════════════════════════════════════════════ // MARKETPULSE — NEWS PAGE v9 // Real NewsAPI + time filters (Today/Week/Month/All) // ═══════════════════════════════════════════════════════ class NewsManager { constructor() { this.allArticles = []; this.category = 'all'; this.timeFilter = 'all'; // today | week | month | all this.sortBy = 'date'; this.query = ''; this.page = 1; this.perPage = 12; this._debounce = null; this.init(); } async init() { await this.fetch('all', ''); } // ── FETCH ──────────────────────────────────────────── async fetch(category = 'all', q = '') { this._showSkeletons(); try { let url = `${window.apiClient.base}/api/news?category=${category}`; if (q) url += `&q=${encodeURIComponent(q)}`; const res = await fetch(url); if (!res.ok) throw new Error(`HTTP ${res.status}`); const articles = await res.json(); if (!Array.isArray(articles)) throw new Error('Bad response'); this.allArticles = articles; this._buildCarousel(); this.render(); } catch(e) { this._showError(e.message); } } // ── CAROUSEL (breaking news — capped height) ───────── _buildCarousel() { const top = this.allArticles.slice(0, 5); const ind = document.getElementById('breakingIndicators'); const inner = document.getElementById('breakingInner'); if (!ind || !inner || !top.length) return; ind.innerHTML = top.map((_,i) => `` ).join(''); inner.innerHTML = top.map((a,i) => `
${a.image ? `` : ''}
BREAKING
${a.title}
${a.source?.toUpperCase()} · ${Fmt.timeAgo(a.published_at)}
`).join(''); } // ── TIME FILTER ─────────────────────────────────────── setTimeFilter(filter, el) { document.querySelectorAll('.time-chip').forEach(c => c.classList.remove('active')); el.classList.add('active'); this.timeFilter = filter; this.page = 1; this.render(); } _filterByTime(articles) { if (this.timeFilter === 'all') return articles; const now = Date.now(); const limits = { today: 24 * 60 * 60 * 1000, week: 7 * 24 * 60 * 60 * 1000, month: 30 * 24 * 60 * 60 * 1000, }; const limit = limits[this.timeFilter]; if (!limit) return articles; return articles.filter(a => { if (!a.published_at) return true; return (now - new Date(a.published_at).getTime()) <= limit; }); } // ── CATEGORY + SORT ─────────────────────────────────── setCategory(cat, el) { document.querySelectorAll('.cat-chip').forEach(c => c.classList.remove('active')); el.classList.add('active'); this.category = cat; this.page = 1; this.query = ''; const si = document.getElementById('newsSearch'); if (si) si.value = ''; document.querySelector('.search-bar')?.classList.remove('has-val'); this.fetch(cat, ''); } setSort(val) { this.sortBy = val; this.render(); } search(q) { this.query = q; this.page = 1; clearTimeout(this._debounce); this._debounce = setTimeout(() => { if (!q) this.fetch(this.category, ''); else if (q.length >= 2) this.fetch('all', q); else this.render(); }, 400); } clearSearch() { this.query = ''; const si = document.getElementById('newsSearch'); if (si) si.value = ''; document.querySelector('.search-bar')?.classList.remove('has-val'); this.fetch(this.category, ''); } // ── RENDER ──────────────────────────────────────────── getFiltered() { let articles = [...this.allArticles]; if (this.query) { const q = this.query.toLowerCase(); articles = articles.filter(a => (a.title||'').toLowerCase().includes(q) || (a.source||'').toLowerCase().includes(q) ); } articles = this._filterByTime(articles); if (this.sortBy === 'source') articles.sort((a,b) => (a.source||'').localeCompare(b.source||'')); else articles.sort((a,b) => new Date(b.published_at||0) - new Date(a.published_at||0)); return articles; } render() { const articles = this.getFiltered(); const paginated = articles.slice(0, this.page * this.perPage); const countEl = document.getElementById('articleCount'); if (countEl) countEl.textContent = `${articles.length} ARTICLES`; const lmWrap = document.getElementById('loadMoreWrap'); if (lmWrap) lmWrap.style.display = articles.length > paginated.length ? 'block' : 'none'; // Show time filter counts this._updateTimeCounts(); const grid = document.getElementById('newsGrid'); if (!grid) return; if (!paginated.length) { grid.innerHTML = `
NO ARTICLES FOUND FOR THIS FILTER
`; return; } grid.innerHTML = paginated.map(a => `
${a.image ? `` : '
📰
'}
${a.source || 'UNKNOWN'} ${Fmt.timeAgo(a.published_at)}
${a.title || 'Untitled'}
${a.description ? `
${a.description.substring(0,120)}...
` : ''}
READ MORE →
`).join(''); setTimeout(() => window.scrollAnimator?.refresh(), 50); } _updateTimeCounts() { const now = Date.now(); const limits = { today: 24*60*60*1000, week: 7*24*60*60*1000, month: 30*24*60*60*1000, }; Object.entries(limits).forEach(([key, ms]) => { const el = document.getElementById(`tc-${key}`); if (!el) return; const count = this.allArticles.filter(a => { if (!a.published_at) return false; return (now - new Date(a.published_at).getTime()) <= ms; }).length; el.textContent = count; }); const allEl = document.getElementById('tc-all'); if (allEl) allEl.textContent = this.allArticles.length; } loadMore() { this.page++; this.render(); Toast.show('MORE ARTICLES LOADED'); } _showSkeletons() { const g = document.getElementById('newsGrid'); if (!g) return; g.innerHTML = Array(12).fill(`
`).join(''); } _showError(msg) { const g = document.getElementById('newsGrid'); if (!g) return; g.innerHTML = `
⚠ BACKEND OFFLINE — ${msg || 'Run: uvicorn main:app --port 8000'}
`; } } document.addEventListener('DOMContentLoaded', () => { window.newsManager = new NewsManager(); }); function setNewsCategory(cat, el) { window.newsManager?.setCategory(cat, el); } function setNewsTimeFilter(f, el) { window.newsManager?.setTimeFilter(f, el); } function setNewsSort(val) { window.newsManager?.setSort(val); } function searchNews(q) { window.newsManager?.search(q); } function clearNewsSearch() { window.newsManager?.clearSearch(); } function loadMoreNews() { window.newsManager?.loadMore(); }
